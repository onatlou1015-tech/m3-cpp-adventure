🎯 Activity 1: About Me

📝 Solution Code:
#include <iostream> #include <string> using namespace std; int main() { // Creating variables with student's own information string name = "Sarah"; int age = 14; string subject = "Science"; int favNum = 7; // Printing the information in a formatted way cout << "===== ABOUT ME =====" << endl; cout << "Name: " << name << endl; cout << "Age: " << age << endl; cout << "Favorite Subject: " << subject << endl; cout << "Favorite Number: " << favNum << endl; cout << "====================" << endl; return 0; }
💡 Expected Output: The program should print all variables in a formatted "About Me" section. Student names and values will vary.

🚀 Activity 2: Simple Calculator

📝 Solution Code:
#include <iostream> using namespace std; int main() { // Variables with chosen numbers int num1 = 15; int num2 = 4; // Calculations int sum = num1 + num2; int difference = num1 - num2; int product = num1 * num2; int quotient = num1 / num2; int remainder = num1 % num2; // Bonus: remainder double average = (num1 + num2) / 2.0; // Bonus: average // Print results cout << "Numbers: " << num1 << " and " << num2 << endl; cout << "Sum: " << sum << endl; cout << "Difference: " << difference << endl; cout << "Product: " << product << endl; cout << "Quotient: " << quotient << endl; cout << "Remainder: " << remainder << endl; cout << "Average: " << average << endl; return 0; }
💡 Expected Output: Numbers: 15 and 4
Sum: 19
Difference: 11
Product: 60
Quotient: 3
Remainder: 3
Average: 9.5

🌟 Bonus Challenge: Custom Greeting

📝 Solution Code:
#include <iostream> #include <string> using namespace std; int main() { // Variables with student's information string firstName = "Alex"; string lastName = "Johnson"; int birthYear = 2009; string hobby = "gaming"; // Calculate age (assuming current year is 2025) int age = 2025 - birthYear; // Print custom greeting cout << "===== WELCOME =====" << endl; cout << "Hello, " << firstName << " " << lastName << "!" << endl; cout << "You were born in " << birthYear << "." << endl; cout << "You are " << age << " years old." << endl; cout << "Your favorite hobby is " << hobby << "." << endl; cout << "Keep coding! 🚀" << endl; return 0; }
💡 Expected Output: ===== WELCOME =====
Hello, Alex Johnson!
You were born in 2009.
You are 16 years old.
Your favorite hobby is gaming.
Keep coding! 🚀

Question 11 - City & Population:
#include <iostream>
#include <string>
using namespace std;

int main() {
    string city = "Toronto";
    int population = 3000000;
    cout << "City: " << city << endl;
    cout << "Population: " << population << endl;
    return 0;
}

Question 12 - Area & Perimeter:
#include <iostream>
using namespace std;

int main() {
    int length = 5;
    int width = 3;
    int area = length * width;
    int perimeter = 2 * length + 2 * width;
    cout << "Length: " << length << endl;
    cout << "Width: " << width << endl;
    cout << "Area: " << area << endl;
    cout << "Perimeter: " << perimeter << endl;
    return 0;
}

Question 18 - Temperature Converter:
#include <iostream>
using namespace std;

int main() {
    double celsius = 25.0;
    double fahrenheit = (celsius * 9.0/5.0) + 32.0;
    cout << "Celsius: " << celsius << "°C" << endl;
    cout << "Fahrenheit: " << fahrenheit << "°F" << endl;
    return 0;
}
